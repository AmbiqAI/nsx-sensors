cmake_minimum_required(VERSION 3.20)
project(sensor_example LANGUAGES C CXX)
if(NSX_SENSORS_SOURCE)
    add_subdirectory("${NSX_SENSORS_SOURCE}" sensors)
else()
    find_package(nsx_sensors CONFIG REQUIRED COMPONENTS bme688 ledstick ina228 mpu6050 max86150 icm45605 tsl254x t5838)
endif()
add_executable(sensor_example main.c)
target_link_libraries(sensor_example PRIVATE nsx::sensor_bme688 nsx::sensor_ledstick nsx::sensor_ina228 nsx::sensor_mpu6050 nsx::sensor_max86150 nsx::sensor_icm45605)
# Compile public headers as C++ and resolve their C linkage as a real consumer.
add_executable(sensor_cpp ../../tests/portable/headers.cpp)
target_link_libraries(sensor_cpp PRIVATE nsx::sensors)
