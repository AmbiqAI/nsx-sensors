if(NSX_SENSOR_BME688)
    add_executable(test_bme688 ../bme688_host_unit/test_nsx_bme688.c)
    target_link_libraries(test_bme688 PRIVATE nsx::sensor_bme688)
    add_test(NAME bme688 COMMAND test_bme688)
endif()
if(NSX_SENSOR_LEDSTICK)
    add_executable(test_bus_ledstick test_bus_ledstick.c)
    target_link_libraries(test_bus_ledstick PRIVATE nsx::sensor_ledstick)
    add_test(NAME bus_ledstick COMMAND test_bus_ledstick)
endif()
if(NSX_SENSOR_INA228)
    add_executable(test_ina228 ../ina228_host_unit/test_nsx_ina228.c)
    target_link_libraries(test_ina228 PRIVATE nsx::sensor_ina228)
    add_test(NAME ina228 COMMAND test_ina228)
endif()
foreach(test test_bme688 test_bus_ledstick test_ina228)
    if(TARGET ${test})
        if(MSVC)
            target_compile_options(${test} PRIVATE /UNDEBUG)
        else()
            target_compile_options(${test} PRIVATE -UNDEBUG -Wall -Wextra -Werror -pedantic)
        endif()
    endif()
endforeach()
