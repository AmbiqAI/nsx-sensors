cmake_minimum_required(VERSION 3.20)
project(nsx_sensor_port_contract LANGUAGES C CXX)
# SDK stand-ins expose the same dependency/export structure without requiring HW.
if(NSX_PORT_INSTALLED)
    include("${NSX_PORT_INSTALLED}/lib/cmake/sdk/fixtureTargets.cmake")
    find_package(nsx_sensors REQUIRED COMPONENTS bme688 ledstick ina228 mpu6050 port_nsx)
else()
    include(GNUInstallDirs)
    foreach(bus i2c spi)
        add_library(fixture_${bus} INTERFACE)
        add_library(nsx::${bus} ALIAS fixture_${bus})
        set_target_properties(fixture_${bus} PROPERTIES EXPORT_NAME ${bus})
        target_include_directories(fixture_${bus} INTERFACE
            $<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/stubs>
            $<INSTALL_INTERFACE:include/sdk>)
        install(TARGETS fixture_${bus} EXPORT fixtureTargets)
    endforeach()
    if(NSX_PORT_PACKAGE_TEST)
        install(EXPORT fixtureTargets NAMESPACE nsx:: DESTINATION lib/cmake/sdk)
        set(NSX_SENSOR_INSTALL_NSX_PORT ON CACHE BOOL "" FORCE)
    endif()
    install(DIRECTORY stubs/ DESTINATION include/sdk)
    set(NSX_SENSOR_PORT_NSX ON CACHE BOOL "" FORCE)
    add_subdirectory(../.. sensors)
endif()
add_executable(port_contract test_port.c)
target_link_libraries(port_contract PRIVATE nsx::sensor_port_nsx nsx::sensor_bme688 nsx::sensor_ledstick nsx::sensor_ina228 nsx::sensor_mpu6050)
add_library(cpp_header OBJECT header.cpp)
target_link_libraries(cpp_header PRIVATE nsx::sensor_port_nsx)
