cmake_minimum_required(VERSION 3.20)
project(nsx_sensor_firmware C ASM)
include(GNUInstallDirs)
if(NOT IS_DIRECTORY "${NSX_SDK_ROOT}")
    message(FATAL_ERROR "NSX_SDK_ROOT must name the pinned SDK checkout")
endif()
if(NOT FIRMWARE_MODE MATCHES "^(baremetal|freertos)$")
    message(FATAL_ERROR "FIRMWARE_MODE must be baremetal or freertos")
endif()
set(NSX_ROOT "${NSX_SDK_ROOT}")
set(NSX_CMAKE_DIR "${NSX_ROOT}/cmake")
set(NSX_SDK_PROVIDER ambiqsuite)
set(NSX_TOOLCHAIN_FAMILY gcc)
set(NSX_AMBIQSUITE_ROOT "${NSX_ROOT}/modules/nsx-ambiqsuite/sdk")
include("${NSX_ROOT}/boards/apollo4p_evb/board.cmake")
foreach(module cmsis-core cmsis-startup ambiq-hal ambiq-bsp soc-hal core i2c spi)
    add_subdirectory("${NSX_ROOT}/modules/nsx-${module}" "sdk/${module}")
endforeach()
target_link_libraries(nsx_startup PUBLIC nsx::runtime)
set(NSX_SENSOR_PORT_NSX ON CACHE BOOL "" FORCE)
add_subdirectory(../.. sensors)
add_executable(firmware main.c)
set_target_properties(firmware PROPERTIES SUFFIX .elf)
target_link_libraries(firmware PRIVATE nsx::board nsx::cmsis_startup
    nsx::ambiq_bsp nsx::soc_hal nsx::sensors nsx::sensor_port_nsx)
if(FIRMWARE_MODE STREQUAL "freertos")
    add_library(freertos_config INTERFACE)
    add_library(nsx::freertos_config ALIAS freertos_config)
    target_include_directories(freertos_config INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
    add_subdirectory("${NSX_ROOT}/modules/nsx-freertos" sdk/freertos)
    target_link_libraries(firmware PRIVATE nsx::freertos)
endif()
target_compile_options(firmware PRIVATE -ffunction-sections -fdata-sections -Wall -Wextra)
target_link_options(firmware PRIVATE -nostartfiles "LINKER:-T,${NSX_LINKER_SCRIPT}"
    "LINKER:-u,Reset_Handler" "LINKER:--gc-sections" "LINKER:-Map,${CMAKE_BINARY_DIR}/firmware.map"
    "LINKER:--wrap=_write_r,--wrap=_read_r,--wrap=_close_r,--wrap=_lseek_r,--wrap=_kill_r,--wrap=_getpid_r,--wrap=_fstat_r,--wrap=_isatty_r")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/sdk-inputs.txt" CONTENT
    "${NSX_STARTUP_SOURCE}\n${NSX_SYSTEM_SOURCE}\n${NSX_LINKER_SCRIPT}\n${NSX_HAL_LIB}\n${NSX_BSP_LIB}\n")
