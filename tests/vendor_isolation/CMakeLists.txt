cmake_minimum_required(VERSION 3.20)
project(vendor_isolation LANGUAGES C CXX)
if(NSX_SENSORS_SOURCE)
    add_subdirectory("${NSX_SENSORS_SOURCE}" nsx-sensors)
else()
    find_package(nsx_sensors CONFIG REQUIRED COMPONENTS icm45605)
endif()
enable_testing()
foreach(language c cpp)
    foreach(order foreign_first nsx_first)
        set(name isolation_${language}_${order})
        add_executable(${name} consumer.${language} foreign.c)
        target_include_directories(${name} PRIVATE foreign)
        target_link_libraries(${name} PRIVATE nsx::sensor_icm45605)
        target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
        if(order STREQUAL "foreign_first")
            target_compile_definitions(${name} PRIVATE FOREIGN_FIRST)
        endif()
        add_test(NAME ${name} COMMAND ${name})
    endforeach()
endforeach()
