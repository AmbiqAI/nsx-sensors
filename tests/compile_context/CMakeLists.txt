cmake_minimum_required(VERSION 3.20)
project(sensor_compile_context C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# Model NSX's transitive target-based ABI settings, without SDK dependencies.
if(NOT OMIT_CONTEXT)
    add_library(profile INTERFACE)
    target_compile_options(profile INTERFACE ${ENUM_FLAG})
    target_compile_definitions(profile INTERFACE SENSOR_CONTEXT_PROBE=1)
    add_library(context INTERFACE)
    target_link_libraries(context INTERFACE profile)
    if(EXPLICIT_CONTEXT)
        add_library(test::context ALIAS context)
        set(NSX_SENSOR_COMPILE_CONTEXT test::context)
    else()
        add_library(nsx::board_flags ALIAS context)
    endif()
endif()
if(SENSOR_SOURCE)
    foreach(bus core i2c spi)
        add_library(sdk_${bus} INTERFACE)
        add_library(nsx::${bus} ALIAS sdk_${bus})
        target_link_libraries(sdk_${bus} INTERFACE context)
        target_include_directories(sdk_${bus} INTERFACE "${CMAKE_CURRENT_LIST_DIR}/../nsx_port/stubs")
    endforeach()
    add_subdirectory("${SENSOR_SOURCE}" sensors)
    # Probe compiled with the actual driver's settings, also shipped in this
    # test-only install to compare a separately compiled package consumer.
    target_sources(nsx_sensor_max86150 PRIVATE "${CMAKE_CURRENT_LIST_DIR}/layout.c")
    add_executable(port_probe ../nsx_port/test_port.c)
    target_link_libraries(port_probe PRIVATE nsx::sensors nsx::sensor_port_nsx)
else()
    find_package(nsx_sensors CONFIG REQUIRED COMPONENTS max86150)
endif()
add_executable(context_probe main.cpp)
target_link_libraries(context_probe PRIVATE nsx::sensor_max86150)
# Model an application with its own framework settings for the original bug.
# The installed consumer only links sensors, verifying profile reattachment.
if(SENSOR_SOURCE AND NOT OMIT_CONTEXT)
    target_link_libraries(context_probe PRIVATE context)
endif()
